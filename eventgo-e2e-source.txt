type CartResult = Awaited<
  ReturnType<typeof getCartAction>
>;

type CartData = NonNullable<CartResult["cart"]>;
type CartItem = CartData["items"][number];

function formatPrice(amount: number, currency: string) {
  return new Intl.NumberFormat("de-DE", {
    style: "currency",
    currency,
  }).format(amount);
}

export default function CartPage() {
  const [cart, setCart] = useState<CartData | null>(null);
  const [loading, setLoading] = useState(true);
  const [updatingId, setUpdatingId] = useState<string | null>(
    null
  );
  const [error, setError] = useState<string | null>(null);

  const loadCart = useCallback(async () => {
    try {
      const response = await getCartAction();
      setCart(response.cart);
      setError(null);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to load your cart."
      );
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    void loadCart();
  }, [loadCart]);

  function notifyCartChanged() {
    window.dispatchEvent(
      new CustomEvent("eventgo:cart-updated")
    );
  }

  async function changeQuantity(
    item: CartItem,
    nextQuantity: number
  ) {
    if (
      updatingId ||
      nextQuantity < 1 ||
      nextQuantity > 10 ||
      nextQuantity > item.available
    ) {
      return;
    }

    setUpdatingId(item.ticketTypeId);
    setError(null);

    try {
      const updated = await updateCartQuantityAction(
        item.ticketTypeId,
        nextQuantity
      );

      setCart(updated);
      notifyCartChanged();
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to update ticket quantity."
      );
    } finally {
      setUpdatingId(null);
    }
  }

  async function removeItem(ticketTypeId: string) {
    if (updatingId) return;

    setUpdatingId(ticketTypeId);
    setError(null);

    try {
      const updated = await removeFromCartAction(
        ticketTypeId
      );

      setCart(updated);
      notifyCartChanged();
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to remove ticket."
      );
    } finally {
      setUpdatingId(null);
    }
  }

  const items = cart?.items ?? [];

  const ticketCount = items.reduce(
    (sum, item) => sum + item.quantity,
    0
  );

  const totals = items.reduce<Record<string, number>>(
    (acc, item) => {
      acc[item.currency] =
        (acc[item.currency] ?? 0) +
        Number(item.unitPrice) * item.quantity;

      return acc;
    },
    {}
  );

        item.available >= item.quantity &&
        item.quantity >= 1 &&
        item.quantity <= 10 &&
        new Date(item.eventDate).getTime() > Date.now()
    );

import { searchEventsAction } from "@/app/actions/event.actions";
import { ExplorePage } from "@/components/ExplorePage/ExplorePage";

import type {
  EventCategory,
  EventFilters,
} from "@/types/event";

interface PageProps {
  searchParams: Promise<{
    city?: string;
    radius?: string;
    date?: string;
    category?: string;
    q?: string;
    page?: string;
  }>;
}

const categories: EventCategory[] = [
  "music",
  "sports",
  "technology",
  "food",
  "art",
  "business",
  "nightlife",
  "festivals",
  "cinema",
  "comedy",
  "culture",
  "wellness",
];

function buildPageUrl(
  params: Record<string, string | undefined>,
  page: number
) {
  const query = new URLSearchParams();

  for (const [key, value] of Object.entries(params)) {
    if (value) {
      query.set(key, value);
    }
  }

  query.set("page", String(page));

  return `/explore?${query.toString()}`;
}

export default async function Page({
  searchParams,
}: PageProps) {
  const params = await searchParams;

  const parsedRadius = Number(params.radius || 25);
  const parsedPage = Number(params.page || 1);

  const filters: EventFilters = {
    city: params.city ?? "",
    radius:
      Number.isFinite(parsedRadius) && parsedRadius > 0
        ? parsedRadius
        : 25,
    date: params.date ?? "",
    category:
      params.category &&
      categories.includes(
        params.category as EventCategory
      )
        ? (params.category as EventCategory)
        : "all",
  };

  const page =
    Number.isSafeInteger(parsedPage) && parsedPage > 0
      ? parsedPage
      : 1;

  const result = await searchEventsAction({
    ...filters,
    q: params.q ?? "",
    page,
    pageSize: 20,
  });

  const urlParams = {
    city: filters.city,
    radius: String(filters.radius),
    date: filters.date,
    category: filters.category,
    q: params.q,
  };

  return (
    <ExplorePage
      filters={filters}
      events={result.events}
      page={result.page}
      hasMore={result.hasMore}
      previousUrl={
        result.page > 1
          ? buildPageUrl(urlParams, result.page - 1)
          : undefined
      }
      nextUrl={
        result.hasMore
          ? buildPageUrl(urlParams, result.page + 1)
          : undefined
      }
    />
  );
}

  try {

      style: "currency",
      currency,
    }).format(value);
  } catch {
    return `${value.toFixed(2)} ${currency}`;

function formatDate(date: string | null) {
  if (!date) return "—";

  return new Intl.DateTimeFormat("de-DE", {
    dateStyle: "medium",
    timeStyle: "short",
    timeZone: "Europe/Berlin",
  }).format(new Date(date));
}

import { Suspense } from "react";

import {
  Wallet,
  Ticket,
  CalendarDays,
  Users,
} from "lucide-react";

import { DashboardStatCard } from "@/components/molecules/DashboardStatCard/DashboardStatCard";
import { DashboardCharts } from "@/components/organisms/DashboardCharts/DashboardCharts";
import { PlatformActivityChart } from "@/components/organisms/dashboard/PlatformActivityChart/PlatformActivityChart";
import { EventCategoriesCard } from "@/components/organisms/dashboard/EventCategoriesCard/EventCategoriesCard";
import { TopPerformingEventsCard } from "@/components/organisms/dashboard/TopPerformingEventsCard/TopPerformingEventsCard";
import { EventParticipantsCard } from "@/components/organisms/dashboard/EventParticipantsCard/EventParticipantsCard";
import { RecentEventsTable } from "@/components/organisms/dashboard/RecentEventsTable/RecentEventsTable";

import { getDashboardDataAction } from "./actions/dashboard.actions";

const categoryColors = [
  "#FF6B53",
  "#4F8EF7",
  "#9B5CF6",
  "#33C26B",
  "#F6C453",
  "#A0AEC0",

const dateFormatter = new Intl.DateTimeFormat("de-DE", {
  dateStyle: "medium",
  timeZone: "Europe/Berlin",
});

const dateTimeFormatter = new Intl.DateTimeFormat("de-DE", {
  dateStyle: "short",
  timeStyle: "short",

const integerFormatter = new Intl.NumberFormat("de-DE");

async function DashboardContent() {
  const {
    metrics,
    dailyData,
    categories,
    topEvents: dbTopEvents,
    participantEvents: dbParticipantEvents,
    recentEvents: dbRecentEvents,
    recentParticipants: dbRecentParticipants,
  } = await getDashboardDataAction();

  // Only confirmed sales and bookings, according
  // to the statuses configured in dashboard.service.ts.
  const stats = [
    {
      label: "Total Revenue",
      value: currencyFormatter.format(metrics.totalRevenue),
      icon: Wallet,
      color: "coral" as const,
    },
    {
      label: "Tickets Sold",
      value: integerFormatter.format(metrics.ticketsSold),
      icon: Ticket,
      color: "blue" as const,
    },
    {
      label: "Events with Sales",
      value: integerFormatter.format(metrics.totalEvents),
      icon: CalendarDays,
      color: "purple" as const,
    },
    {
      label: "Customers",
      value: integerFormatter.format(metrics.totalUsers),
      icon: Users,
      color: "green" as const,
    },
  ];

  const categoryData = categories.map((category, index) => ({
    name: category.name,
    value: category.value,
    color: categoryColors[index % categoryColors.length],
  }));

  const topEvents = dbTopEvents.map((event, index) => ({
    rank: index + 1,
    title: event.title,
    location: event.city,
    ticketsSold: event.ticketsSold,
  }));

  // Unique customers per event.
  const participantEvents = dbParticipantEvents.map(
    (event, index) => ({
      rank: index + 1,
      title: event.title,
      location: event.city,
      count: event.ticketsSold,
    })
  );

  const recentEvents = dbRecentEvents.map((event) => ({
    id: event.id,
    title: event.title,
    city: event.city,
    category: event.category,
    date: dateFormatter.format(new Date(event.startsAt)),
    tickets: `${event.ticketsSold} / ${event.capacity}`,
    status: event.isDemo ? "Demo" : "Real",
  }));

  const recentParticipants = dbRecentParticipants.map(
    (participant) => ({
      id: participant.id,
      name: participant.name,
      email: participant.email,
      eventTitle: participant.eventTitle,
      timeAgo: participant.purchasedAt
        ? dateTimeFormatter.format(
            new Date(participant.purchasedAt)
          )
        : "—",
    })
  );

  // Normalize revenue and tickets separately.
  // This is a relative activity comparison, not raw amounts.
  const maxRevenue = Math.max(
    1,
    ...dailyData.map((item) => item.revenue)
  );

  const maxTickets = Math.max(
    1,
    ...dailyData.map((item) => item.tickets)
  );

  const platformActivityData = dailyData.map((item) => ({
    date: item.date.slice(5),
    revenue: Math.round(
      (item.revenue / maxRevenue) * 100
    ),
    tickets: Math.round(
      (item.tickets / maxTickets) * 100
    ),
  }));

  return (
    <div className="w-full space-y-7 2xl:space-y-9">
      {/* DASHBOARD HEADER */}
      <section>
        <h1 className="text-3xl font-extrabold tracking-tight text-[#08111D] 2xl:text-4xl">
          Dashboard Overview
        </h1>

        <p className="mt-2 text-sm text-slate-500 2xl:text-base">
          Sales, bookings and participation across your events.
        </p>
      </section>

      {/* METRICS */}
      <section className="grid grid-cols-1 gap-5 sm:grid-cols-2 xl:grid-cols-4 2xl:gap-7">
        {stats.map((stat) => (
          <DashboardStatCard
            key={stat.label}
            {...stat}
          />
        ))}
      </section>

      {/* REVENUE AND TICKET SALES */}
      <section className="min-w-0">
        <DashboardCharts data={dailyData} />
      </section>

      {/* FULL WIDTH PLATFORM ACTIVITY */}
      <section className="min-w-0">
        <PlatformActivityChart
          data={platformActivityData}
        />
      </section>

      {/* CATEGORIES AND TOP EVENTS */}
      <section className="grid grid-cols-1 items-stretch gap-6 lg:grid-cols-2 2xl:grid-cols-3">
        <EventCategoriesCard
          data={categoryData}
          totalEvents={metrics.totalEvents}
        />

        <TopPerformingEventsCard
          items={topEvents}
        />

        <EventParticipantsCard
          items={participantEvents}
        />
      </section>

      {/* RECENT ACTIVITY */}
      <section className="grid grid-cols-1 items-stretch gap-6 2xl:grid-cols-3">
        <div className="min-w-0 2xl:col-span-2">
          <RecentEventsTable items={recentEvents} />
        </div>

        <div className="min-w-0">
          <RecentParticipantsCard
            items={recentParticipants}
          />
        </div>
      </section>
    </div>
  );
}
